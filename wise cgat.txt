    private async Task OpenChat(string chatId)
    {
        try
        {
            var response = await HttpClient.GetAsync($"api/Chat/{chatId}");
            if (response.IsSuccessStatusCode)
            {
                var json = await response.Content.ReadAsStringAsync();
                var chat = System.Text.Json.JsonSerializer.Deserialize<Chat>(json, new JsonSerializerOptions
                {
                    PropertyNamingPolicy = JsonNamingPolicy.CamelCase
                });

                if (chat != null)
                {
                    _discourse.Clear();
                    foreach (var message in chat.Messages)
                    {
                        var conversation = new Conversation
                        {
                            ActorName = message.Sender == "user" ? "You" : "Wise-AI",
                            ActorType = message.Sender == "user" ? ActorType.Self : ActorType.AiBot,
                            Message = message.Content,
                            Time = message.Timestamp
                        };

                        if (message.ContentType == "chart")
                            conversation.ChartData = message.ChartData;
                        else if (message.ContentType == "table")
                            conversation.TableData = message.TableData;
                        else if (message.ContentType == "barChart")
                            conversation.BarChartData = message.BarChartData;
                        else if (message.ContentType == "actDialog")
                        {
                            conversation.ActDialogActions = message.ActDialogData?.Actions;
                            conversation.ActDialogComponent = message.ActDialogData?.Component;
                        }

                        _discourse.Add(conversation);
                    }
                    _hasFirstQuery = true;
                    _currentLayout = "chat";
                    _currentChatId = chat.ChatId;
                    await InvokeAsync(StateHasChanged);
                    await ScrollToBottom();
                }
            }
            else
            {
                Logger.LogWarning("Failed to open chat {ChatId}: {StatusCode}", chatId, response.StatusCode);
                Snackbar.Add("Failed to open chat", Severity.Warning);
            }
        }
        catch (Exception ex)
        {
            Logger.LogError(ex, "Error opening chat {ChatId}", chatId);
            Snackbar.Add($"Error opening chat: {ex.Message}", Severity.Error);
        }
    }

            _currentMessage = string.Empty; // Clear the input field
            await InvokeAsync(StateHasChanged);
            await _refEntryField!.BlurAsync();

            // Save after user message
            await SaveCurrentChat();

            try
            {
                var payload = new
                {
                    query = userQuery,
                    app_context = SelectedAppContext,
                    organization_uuid = _organizationId
                };

                var stringContent = new StringContent(JsonConvert.SerializeObject(payload), System.Text.Encoding.UTF8, "application/json");
                var routerResponse = await HttpClient.PostAsync("http://40.76.230.77:8892/query-router", stringContent);

                if (!routerResponse.IsSuccessStatusCode)
                {
                    throw new HttpRequestException($"Error: Received status code {routerResponse.StatusCode} from query-router endpoint.");
                }

                var routerJson = await routerResponse.Content.ReadAsStringAsync();
                var routerResult = JsonConvert.DeserializeObject<List<dynamic>>(routerJson);
                var route = routerResult[0].name.ToString();
                var value = routerResult[0].value.ToString();

                string response;

                if (route == "function_router")
                {
                    try
                    {
                        var app_context = routerResult[0].app_context.ToString();
                        var queryPayload = new List<object>
                        {
                             new { value = value, app_context = app_context, organization_uuid = _organizationId }
                        };
                        var functionResponse = await HttpClient.PostAsJsonAsync("http://40.76.230.77:8891/execute_queries", queryPayload);

                        if (!functionResponse.IsSuccessStatusCode)
                        {
                            throw new HttpRequestException($"Error: Received status code {functionResponse.StatusCode} from function_router endpoint.");
                        }

                        var responseContent = await functionResponse.Content.ReadAsStringAsync();
                        var jsonResponse = JsonConvert.DeserializeObject<FunctionRouterResponse>(responseContent);

                        if (jsonResponse?.Results != null && jsonResponse.Results.Count > 0)
                        {
                            var chartDataJson = jsonResponse.Results[0];
                            var rawChartData = JsonConvert.DeserializeObject<dynamic>(chartDataJson);


                            var combinedContent = $"Don't do any further analysis just create a 1 line summary.Use User Query and Response data to get only one line summary of the Response Data based on the User Query. User Query: {userQuery}\n\nResponse Data: {responseContent}";

                            if (rawChartData.type == "pie_chart")
                            {
                                var chartData = new ChartData();
                                foreach (var chart in rawChartData.data)
                                {
                                    chartData.Charts.Add(new Chart
                                    {
                                        ChartName = chart.chart_name,
                                        InputData = ((JArray)chart.InputData).ToObject<List<int>>(),
                                        InputLabels = ((JArray)chart.InputLabels).ToObject<List<string>>()
                                    });
                                }
                                response = await WiseAI.GetResponseAsync(_wiseAIThreadId, combinedContent);
                                await AddAiContentToChat(response, chartData);
                            }
                            else if (rawChartData.type == "table")
                            {
                                var tableData = new TableData
                                {
                                    Headings = ((JArray)rawChartData.data.headings).ToObject<List<string>>(),
                                    Rows = ((JArray)rawChartData.data.rows).ToObject<List<List<string>>>()
                                };
                                response = await WiseAI.GetResponseAsync(_wiseAIThreadId, combinedContent);
                                await AddAiContentToChat(response, tableData: tableData);
                            }
                            else if (rawChartData.type == "bar_chart")
                            {
                                var barChartData = new BarChartData
                                {
                                    Hours = ((JArray)rawChartData.data[0].hours).ToObject<List<string>>(),
                                    Counts = ((JArray)rawChartData.data[0].counts).ToObject<List<int>>()
                                };
                                response = await WiseAI.GetResponseAsync(_wiseAIThreadId, combinedContent);
                                await AddAiContentToChat(response, barChartData: barChartData);
                            }
                            else if (rawChartData.type == "act_dialog")
                            {
                                var actDialogData = JsonConvert.DeserializeObject<ActDialogResponse>(chartDataJson);
                                var actions = new List<ActionResolution>();
                                for (int i = 0; i < actDialogData.data.issues.Count; i++)
                                {
                                    var issue = actDialogData.data.issues[i];
                                    actions.Add(new ActionResolution
                                    {
                                        Description = issue.description,
                                        Resolutions = issue.action_steps ?? new List<string>(),
                                        IssueId = i + 1,
                                        ActionId = i + 1
                                    });
                                }
                                var component = new Component { CId = actDialogData.data.component_id };
                                response = await WiseAI.GetResponseAsync(_wiseAIThreadId, combinedContent);
                                await AddAiContentToChat(response, actDialogActions: actions, actDialogComponent: component);
                            }
                            else
                            {
                                response = rawChartData.ToString();
                                await AddAiContentToChat(response);
                            }
                        }
                        else
                        {
                            response = "Unexpected JSON structure in the response.";
                            await AddAiContentToChat(response);
                        }
                    }
                    catch (Exception ex)
                    {
                        response = $"An error occurred while processing the data: {ex.Message}";

                        _discourse.Add(new Conversation
                        {
                            ActorName = "Wise-AI",
                            ActorType = ActorType.AiBot,
                            Message = response,
                            Time = DateTime.Now,
                            IsError = true
                        });
                    }
                }
                else
                {
                    response = await WiseAI.GetResponseAsync(_wiseAIThreadId, value);
                    await AddAiContentToChat(response);
                }

                // Save after AI response
                await SaveCurrentChat();
            }
            catch (HttpRequestException ex)
            {

                {
                    ActorName = "Wise-AI",
                    ActorType = ActorType.AiBot,
                    Message = errorMessage,
                    Time = DateTime.Now,

                {
                    ActorName = "Wise-AI",
                    ActorType = ActorType.AiBot,
                    Message = errorMessage,
                    Time = DateTime.Now,

            }
            finally
            {
                if (_isLoading)

                    _isLoading = false;
                    await InvokeAsync(StateHasChanged);
                }

                {
                    await _refEntryField!.Clear();

            }
        }
    }

    private string GetRelativeTime(DateTime dateTime)
    {
        // Convert the UTC time from the database to the server's local time
        var localTime = dateTime.ToLocalTime();
        var timeSpan = DateTime.Now - localTime;

        if (timeSpan.TotalMinutes < 1) return "Just now";
        if (timeSpan.TotalMinutes < 60) return $"{(int)timeSpan.TotalMinutes} minutes ago";
        if (timeSpan.TotalHours < 24) return $"{(int)timeSpan.TotalHours} hours ago";
        if (timeSpan.TotalDays < 2) return "Yesterday";

        // For older dates, show the local time
        return localTime.ToString("h:mm tt");
    }

        {

            _httpClient.DefaultRequestHeaders.Add("OpenAI-Beta", "assistants=v2");

        public async Task<string> Initialized(Agent agent)
        {
            try
            {
                Console.WriteLine($"Fetching assistant with ID: {agent.AssistantId}");

                // Fetch the assistant
                var response = await _httpClient.GetAsync($"assistants/{agent.AssistantId}");
                Console.WriteLine($"Assistant fetch response status: {response.StatusCode}");

                if (!response.IsSuccessStatusCode)
                {
                    var errorContent = await response.Content.ReadAsStringAsync();
                    Console.WriteLine($"Assistant fetch error: {errorContent}");
                    return $"Error fetching assistant: {errorContent}";
                }

                _assistant = await response.Content.ReadFromJsonAsync<AssistantResponse>();
                Console.WriteLine($"Assistant fetched successfully: {_assistant?.Id}");

                // Create a new thread
                Console.WriteLine("Creating a new thread...");
                var threadResponse = await _httpClient.PostAsync("threads", new StringContent("{}", Encoding.UTF8, "application/json"));
                Console.WriteLine($"Thread creation response status: {threadResponse.StatusCode}");

                if (!threadResponse.IsSuccessStatusCode)
                {
                    var errorContent = await threadResponse.Content.ReadAsStringAsync();
                    Console.WriteLine($"Thread creation error: {errorContent}");
                    return $"Error creating thread: {errorContent}";
                }

                var thread = await threadResponse.Content.ReadFromJsonAsync<ThreadResponse>();

                if (thread != null)
                {
                    Console.WriteLine($"Thread created successfully: {thread.Id}");
                    _threads.Add(thread.Id, thread); // Ensure the thread is added to the dictionary
                    return thread.Id;
                }

                Console.WriteLine("Error: Thread creation failed.");
                return "Error creating thread";
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Error initializing AI session: {ex.Message}");
                return "Error Initializing AI Session";
            }
        }


        public async Task<string> RunAndGetResponse(string threadId, string userPrompt)
        {
            try
            {
                Console.WriteLine($"Running chat for thread ID: {threadId}");

                if (!_threads.ContainsKey(threadId))
                {
                    Console.WriteLine("Error: Invalid thread ID");
                    return "Invalid thread ID";
                }

                // Send user message
                Console.WriteLine("Sending user message...");
                var messageContent = new { role = "user", content = userPrompt };
                var jsonMessageContent = JsonSerializer.Serialize(messageContent);
                Console.WriteLine($"Sending JSON payload: {jsonMessageContent}");
                var messageResponse = await _httpClient.PostAsync($"threads/{threadId}/messages", new StringContent(jsonMessageContent, Encoding.UTF8, "application/json"));
                Console.WriteLine($"Message response status: {messageResponse.StatusCode}");

                if (!messageResponse.IsSuccessStatusCode)
                {
                    var errorContent = await messageResponse.Content.ReadAsStringAsync();
                    Console.WriteLine($"Message error: {errorContent}");
                    return $"Error sending message: {errorContent}";
                }

                // Create a run
                Console.WriteLine("Creating a run...");
                var runContent = new { assistant_id = "asst_cHJIUvqUoB8lH7z61bZV3cnq" };
                var jsonRunContent = JsonSerializer.Serialize(runContent);
                Console.WriteLine($"Sending JSON payload: {jsonRunContent}");
                var runResponse = await _httpClient.PostAsync($"threads/{threadId}/runs", new StringContent(jsonRunContent, Encoding.UTF8, "application/json"));
                Console.WriteLine($"Run response status: {runResponse.StatusCode}");

                if (!runResponse.IsSuccessStatusCode)
                {
                    var errorContent = await runResponse.Content.ReadAsStringAsync();
                    Console.WriteLine($"Run error: {errorContent}");
                    return $"Error creating run: {errorContent}";
                }

                var rawJson = await runResponse.Content.ReadAsStringAsync();
                Console.WriteLine($"Raw run response: {rawJson}");
                var options = new JsonSerializerOptions { PropertyNameCaseInsensitive = true };
                var run = JsonSerializer.Deserialize<RunResponse>(rawJson, options);

                if (run == null || string.IsNullOrEmpty(run.Id))
                {
                    Console.WriteLine("Error: Run object is null or ID is empty.");
                    return "Error: Run creation failed.";
                }

                Console.WriteLine($"Run created successfully: {run.Id}, Status: {run.Status}");

                // Poll run status
                while (run.Status == "in_progress" || run.Status == "queued" || run.Status == "cancelling")
                {
                    Console.WriteLine($"Run status: {run.Status}");
                    await Task.Delay(250);
                    var statusResponse = await _httpClient.GetAsync($"threads/{threadId}/runs/{run.Id}");
                    rawJson = await statusResponse.Content.ReadAsStringAsync();
                    Console.WriteLine($"Raw status response: {rawJson}");
                    run = JsonSerializer.Deserialize<RunResponse>(rawJson, options);
                }

                Console.WriteLine($"Run completed with status: {run.Status}");

                if (run.Status != "completed")
                {
                    Console.WriteLine($"Run failed with status: {run.Status}");
                    return $"Run failed with status: {run.Status}";
                }

                // Fetch messages
                Console.WriteLine("Fetching messages...");
                var messagesResponse = await _httpClient.GetAsync($"threads/{threadId}/messages");
                Console.WriteLine($"Messages response status: {messagesResponse.StatusCode}");

                if (!messagesResponse.IsSuccessStatusCode)
                {
                    var errorContent = await messagesResponse.Content.ReadAsStringAsync();
                    Console.WriteLine($"Messages error: {errorContent}");
                    return $"Error fetching messages: {errorContent}";
                }

                var messagesRawJson = await messagesResponse.Content.ReadAsStringAsync();
                Console.WriteLine($"Raw messages response: {messagesRawJson}");
                var messageList = JsonSerializer.Deserialize<MessageListResponse>(messagesRawJson, options);

                if (messageList?.Items.Count > 0) // Check for any messages, not > 1
                {
                    var assistantMessage = messageList.Items.FirstOrDefault(m => m.Role == "assistant");
                    if (assistantMessage != null)
                    {
                        var response = assistantMessage.PrintContent();
                        Console.WriteLine($"Assistant response: {response}");
                        return response;
                    }
                    Console.WriteLine("No assistant message found");
                    return "No assistant message found";
                }

                Console.WriteLine("No messages in the thread");
                return "No messages in the thread";
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Error retrieving AI response: {ex.Message}");
                return $"Error retrieving AI response: {ex.Message}";
            }
        }
        public async Task<List<string>> GenerateRecommendedQuestions(string threadId)
        {
            try
            {
                if (!_threads.ContainsKey(threadId))
                {
                    return new List<string>();
                }
                var prompt = "Analyze the conversation history from the user's perspective. Generate 3 diverse, highly relevant follow-up questions the user would likely ask next, based on their interests, needs, and the conversation context. Ensure a balanced mix of: 1. Data insight questions (only when contextually appropriate) such as: 'At what time did the most incidents occur on a specific date?', 'What's the overall health of all my systems in a graph?', 'What systems have high/low CPU usage?', 'Which systems have low memory/disk usage?', 'What systems show the 'Data Metric Missing' error?'. 2. General product usage or feature-related questions (e.g., 'How do I set up a new alert?', 'Can I customize the dashboard layout?'). 3. Ensure all questions: Are in first-person, sound natural for the user to ask, follow the conversation flow, seek valuable information, and avoid using 'visual' or 'visualization'. Return only the questions as a JSON array, without additional text or formatting. Each question should appear to come directly from the user to the AI assistant.";

                var runContent = new { assistant_id = _assistant.Id };

                while (run.Status == "in_progress" || run.Status == "queued")

                    await Task.Delay(250);

                    run = await statusResponse.Content.ReadFromJsonAsync<RunResponse>();
                }

                var messageResponse = await _httpClient.GetAsync($"threads/{threadId}/messages");
                var messageList = await messageResponse.Content.ReadFromJsonAsync<MessageListResponse>();

                if (messageList?.Items.Count > 0)
                {
                    var response = messageList.Items.First().PrintContent();
                    return ParseRecommendations(response);
                }
                return new List<string>();
            }
            catch (Exception ex)
            {
                return new List<string>();
            }
        }
        private List<string> ParseRecommendations(string response)
        {
            try
            {
                return JsonSerializer.Deserialize<List<string>>(response) ?? new List<string>();
            }
            catch (JsonException)
            {
                return response.Split('\n')
                    .Where(line => line.Trim().StartsWith("1.") || line.Trim().StartsWith("2.") || line.Trim().StartsWith("3."))
                    .Select(line => line.Substring(line.IndexOf('.') + 1).Trim())
                    .ToList();
            }
        }

        public async Task SaveChatAsync(Chat chat)
        {
            if (string.IsNullOrEmpty(chat.ChatId))
            {
                chat.ChatId = Guid.NewGuid().ToString();
                chat.CreatedAt = DateTime.UtcNow;
                await _chatsCollection.InsertOneAsync(chat);
            }
            else
            {
                chat.UpdatedAt = DateTime.UtcNow; // Set only on update
                var filter = Builders<Chat>.Filter.Eq(c => c.ChatId, chat.ChatId);
                await _chatsCollection.ReplaceOneAsync(filter, chat, new ReplaceOptions { IsUpsert = true });
            }
        }

        public async Task<List<Chat>> GetRecentChatAsync(string userId)

            var filter = Builders<Chat>.Filter.Eq(c => c.UserId, userId);
            var sort = Builders<Chat>.Sort.Descending(c => c.UpdatedAt != null ? c.UpdatedAt : c.CreatedAt);
            return await _chatsCollection.Find(filter).Sort(sort).ToListAsync(); // Limit to recent 20
        }

        public async Task<Chat> GetChatByIdAsync(string chatId)
        {
            var filter = Builders<Chat>.Filter.Eq(c => c.ChatId, chatId);
            return await _chatsCollection.Find(filter).FirstOrDefaultAsync();
        }

        public async Task DeleteChatAsync(string chatId)
        {
            var filter = Builders<Chat>.Filter.Eq(c => c.ChatId, chatId);
            await _chatsCollection.DeleteOneAsync(filter);
        }

using Microsoft.AspNetCore.Mvc;
using Wise_Ultima.Models;
using Wise_Ultima.Services;
using System.Threading.Tasks;
using System.Collections.Generic;
using System.ComponentModel.DataAnnotations;
using Microsoft.AspNetCore.Authorization;

namespace Wise_Ultima.Controllers
{
   
    [ApiController]
    [Route("api/[controller]")]
   
    public class WiseAIController : ControllerBase
    {
        private readonly WiseAI _wiseAI;

        public WiseAIController(WiseAI wiseAI)
        {
            _wiseAI = wiseAI ?? throw new ArgumentNullException(nameof(wiseAI));
        }

      
        [HttpPost("initialize")]
        public async Task<IActionResult> Initialize([FromBody][Required] Agent agent)
        {
            if (!ModelState.IsValid || string.IsNullOrEmpty(agent?.AssistantId))
            {
                return BadRequest(new { error = "Invalid or missing assistant ID" });
            }

            var threadId = await _wiseAI.Initialized(agent);
            if (threadId.StartsWith("Error"))
            {
                return StatusCode(500, new { error = threadId });
            }

            return Ok(new { threadId });
        }

     
        [HttpPost("chat")]
        public async Task<IActionResult> GetChatResponse([FromBody][Required] ChatRequest request)
        {
            if (!ModelState.IsValid || string.IsNullOrEmpty(request?.ThreadId) || string.IsNullOrEmpty(request?.UserPrompt))
            {
                return BadRequest(new { error = "Thread ID and user prompt are required" });
            }

            var response = await _wiseAI.RunAndGetResponse(request.ThreadId, request.UserPrompt);
            if (response.StartsWith("Error") || response.Contains("Invalid") || response.Contains("No assistant message") || response.Contains("No messages"))
            {
                return BadRequest(new { error = response });
            }

            return Ok(new { response });
        }

        
        [HttpGet("recommended-questions/{threadId}")]
        public async Task<IActionResult> GetRecommendedQuestions([FromRoute] string threadId)
        {
            if (string.IsNullOrEmpty(threadId))
            {
                return BadRequest(new { error = "Thread ID is required" });
            }

            var questions = await _wiseAI.GenerateRecommendedQuestions(threadId);
            if (questions == null || questions.Count == 0)
            {
                return Ok(new { questions = new List<string>() });
            }

            return Ok(new { questions });
        }

      
        [HttpPost("chat/save")]
        public async Task<IActionResult> SaveChat([FromBody][Required] Chat chat)
        {
            if (!ModelState.IsValid || string.IsNullOrEmpty(chat?.UserId))
            {
                return BadRequest(new { error = "Invalid chat data or missing user ID" });
            }

            // Ensure UserId matches authenticated user (if applicable)
            var currentUserId = User?.Identity?.Name; // Adjust based on auth mechanism
            if (!string.IsNullOrEmpty(currentUserId) && chat.UserId != currentUserId)
            {
                return Unauthorized(new { error = "Cannot save chat for another user" });
            }

            await _wiseAI.SaveChatAsync(chat);
            return Ok(new { chatId = chat.ChatId, message = "Chat saved successfully" });
        }

      
        [HttpGet("chats/recent")]
        public async Task<IActionResult> GetRecentChats()
        {
            var userId = User?.Identity?.Name; // Adjust based on auth mechanism
            if (string.IsNullOrEmpty(userId))
            {
                return Unauthorized(new { error = "User authentication required" });
            }

            var chats = await _wiseAI.GetRecentChatAsync(userId);
            return Ok(new { chats });
        }


        [HttpGet("chat/{chatId}")]
        public async Task<IActionResult> GetChatById([FromRoute] string chatId)
        {
            if (string.IsNullOrEmpty(chatId))
            {
                return BadRequest(new { error = "Chat ID is required" });
            }

            var chat = await _wiseAI.GetChatByIdAsync(chatId);
            if (chat == null)
            {
                return NotFound(new { error = "Chat not found" });
            }

            // Verify user ownership
            var currentUserId = User?.Identity?.Name;
            if (!string.IsNullOrEmpty(currentUserId) && chat.UserId != currentUserId)
            {
                return Unauthorized(new { error = "Access to chat denied" });
            }

            return Ok(new { chat });
        }

        /// <summary>
        /// Deletes a specific chat by ID.
        /// </summary>
        /// <param name="chatId">The ID of the chat to delete.</param>
        /// <returns>Success message.</returns>
        [HttpDelete("chat/{chatId}")]
        public async Task<IActionResult> DeleteChat([FromRoute] string chatId)
        {
            if (string.IsNullOrEmpty(chatId))
            {
                return BadRequest(new { error = "Chat ID is required" });
            }

            var chat = await _wiseAI.GetChatByIdAsync(chatId);
            if (chat == null)
            {
                return NotFound(new { error = "Chat not found" });
            }

            // Verify user ownership
            var currentUserId = User?.Identity?.Name;
            if (!string.IsNullOrEmpty(currentUserId) && chat.UserId != currentUserId)
            {
                return Unauthorized(new { error = "Cannot delete another user's chat" });
            }

            await _wiseAI.DeleteChatAsync(chatId);
            return Ok(new { message = "Chat deleted successfully" });
        }
    }

    /// <summary>
    /// Request model for chat interactions.
    /// </summary>
    public class ChatRequest
    {
        [Required]
        public string ThreadId { get; set; }
        [Required]
        public string UserPrompt { get; set; }
    }
