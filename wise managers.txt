@code {
    private MudTable<PaginatedUser> _table;
    private string _searchString = "";
    private SortDirection _sortDirection = SortDirection.Ascending;
    private bool _loading = false;
    private const string DEFAULT_ROLE = "User";

    // Removed _userRoles, _isInitialized, OnInitializedAsync (trigger handles defaults)

    private string GetFormattedUsername(string email)
    {
        if (string.IsNullOrEmpty(email) || !email.Contains('@'))
            return email;

        string username = email.Split('@')[0];
        if (string.IsNullOrEmpty(username))
            return email;

        return string.Join(" ", username.Split('.')
            .Select(part => !string.IsNullOrEmpty(part)
                ? char.ToUpper(part[0]) + (part.Length > 1 ? part.Substring(1) : "")
                : ""));
    }

  
  
        private async Task<TableData<PaginatedUser>> ServerReload(TableState state)
        {
            try
            {
                _loading = true;
                StateHasChanged();
                _sortDirection = state.SortDirection;
                var search = string.IsNullOrEmpty(_searchString) ? "" : _searchString;
                var (users, totalCount) = await AuthService.GetPaginatedUsers(state.Page + 1, state.PageSize, search, state.SortLabel ?? "email", _sortDirection == SortDirection.Ascending);
                Logger.LogInformation($"Fetched {users.Count} users, TotalCount: {totalCount}");
                return new TableData<PaginatedUser> { TotalItems = totalCount, Items = users };
            }
            catch (Exception ex)
            {
                Logger.LogError(ex, "Error loading users");
                Snackbar.Add($"Failed to load users: {ex.Message}", Severity.Error);
                return new TableData<PaginatedUser> { TotalItems = 0, Items = new List<PaginatedUser>() };
            }
            finally
            {
                _loading = false;
                StateHasChanged();
            }
        }
    
    private async Task OpenEditRolesDialog(PaginatedUser user)
    {
        try
        {
            var allRoles = new List<string> { "Admin", "User", "Manager", "Partner", "Client" };
            var userRole = user.Role ?? DEFAULT_ROLE;

            var parameters = new DialogParameters
            {
                ["User"] = user,
                ["AllRoles"] = allRoles,
                ["UserRole"] = userRole
            };

            var options = new DialogOptions { CloseButton = true, MaxWidth = MaxWidth.Small, FullWidth = true };
            var dialog = await DialogService.ShowAsync<EditUserRoleDialog>("Edit User Role", parameters, options);
            var result = await dialog.Result;

            if (!result.Canceled && result.Data is string newRole)
            {
                var success = await AuthService.UpdateUserRole(user.Id, newRole);
                if (string.IsNullOrEmpty(newRole))
                {
                    newRole = DEFAULT_ROLE;
                }

                // Get current user details for logging
                string currentUserId = "unknown";
                string currentUserName = "unknown";
                try
                {
                    var (userId, email, _) = await AuthService.GetUserDetails();
                    currentUserId = userId ?? "unknown";
                    currentUserName = GetFormattedUsername(email ?? "unknown@example.com");
                }
                catch (Exception ex)
                {
                    Logger.LogWarning(ex, "Failed to get current user details for audit logging");
                }

                var logEntry = new AuditLog
                {
                    UserId = currentUserId,
                    UserName = currentUserName,
                    Action = $"Updated role for user {GetFormattedUsername(user.Email)} to {newRole}",
                    ActionCategory = "User Management",
                    Details = $"User ID: {user.Id}, New Role: {newRole}",
                    Success = success,
                    ErrorMessage = success ? null : "Role update failed",
                    Timestamp = DateTime.UtcNow
                };

                try
                {
                    await AuditLogService.LogActionAsync(logEntry);
                }
                catch (Exception ex)
                {
                    Logger.LogError(ex, "Failed to log role update action for user {UserEmail}", user.Email);
                    Snackbar.Add("Role updated but audit log failed to save.", Severity.Warning);
                }

                if (success)
                {
                    user.Role = newRole;
                    Snackbar.Add($"Role updated for {GetFormattedUsername(user.Email)}", Severity.Success);
                    await _table.ReloadServerData();
                }
                else
                {
                    Logger.LogError("Failed to update role for {UserEmail}", user.Email);
                    Snackbar.Add("Failed to update user role. Please try again.", Severity.Error);
                }
            }
        }
        catch (Exception ex)
        {
            Logger.LogError(ex, "Error in OpenEditRolesDialog for user {UserEmail}", user.Email);
            Snackbar.Add("An error occurred while editing user role. Please try again.", Severity.Error);
        }
    }

    private async Task DeleteUser(PaginatedUser user)
    {
        try
        {
            var success = await AuthService.DeleteUser(user.Id);

            // Get current user details for logging
            string currentUserId = "unknown";
            string currentUserName = "unknown";
            try
            {
                var (userId, email, _) = await AuthService.GetUserDetails();
                currentUserId = userId ?? "unknown";
                currentUserName = GetFormattedUsername(email ?? "unknown@example.com");
            }
            catch (Exception ex)
            {
                Logger.LogWarning(ex, "Failed to get current user details for audit logging");
            }

            var logEntry = new AuditLog
            {
                UserId = currentUserId,
                UserName = currentUserName,
                Action = $"Deleted user {GetFormattedUsername(user.Email)}",
                ActionCategory = "User Management",
                Details = $"Deleted a user email {user.Email}",
                Success = success,
                ErrorMessage = success ? null : "User deletion failed",
                Timestamp = DateTime.UtcNow
            };

            try
            {
                await AuditLogService.LogActionAsync(logEntry);
            }
            catch (Exception ex)
            {
                Logger.LogError(ex, "Failed to log delete action for user {UserEmail}", user.Email);
                Snackbar.Add("User action completed but audit log failed to save.", Severity.Warning);
            }

            if (success)
            {
                Snackbar.Add($"User {GetFormattedUsername(user.Email)} deleted successfully", Severity.Success);
                await _table.ReloadServerData();
            }
            else
            {
                Logger.LogError($"Failed to delete user {user.Id}");
                Snackbar.Add("Failed to delete user.", Severity.Error);
            }
        }
        catch (Exception ex)
        {
            Logger.LogError(ex, "Error deleting user {UserEmail}", user.Email);
            Snackbar.Add("An error occurred while deleting the user. Please try again.", Severity.Error);
        }
    }

    private async Task ConfirmDeleteUser(PaginatedUser user)
    {
        var parameters = new DialogParameters
        {
            ["ContentText"] = $"Are you sure you want to delete the user '{GetFormattedUsername(user.Email)}'?",
            ["ButtonText"] = "Delete",
            ["Color"] = Color.Error
        };

        var options = new DialogOptions { CloseButton = true, MaxWidth = MaxWidth.ExtraSmall, FullWidth = true };
        var dialog = await DialogService.ShowAsync<ConfirmDialog>("Delete User", parameters, options);
        var result = await dialog.Result;

        if (!result.Canceled)
        {
            await DeleteUser(user);
        }
    }

    {
        if (firstRender)
        {
            await _table.ReloadServerData();
        }
    }

    {
        // Clean up resources if necessary
    }
