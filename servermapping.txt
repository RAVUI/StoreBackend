    private async Task OnTierSelected(TierData newTier)
    {
        if (newTier == null || newTier.InstanceType == selectedTier) return;

        isTooltipVisible = false;

        tierInsightTooltip = string.Empty;

        if (!string.IsNullOrEmpty(recommendedRegion) && availableRegions.Contains(recommendedRegion))

    private void UpdateServerDetails()
    {

    private string GetComparisonStyle(decimal? currentValue, decimal? comparisonValue)
    {
        if (currentValue == null || comparisonValue == null)
            return "color: #e2e8f0;";

        if (currentValue > comparisonValue)
            return "color: #10b981;"; // Green for increase
        else if (currentValue < comparisonValue)
            return "color: #f87171;"; // Red for decrease
        else
            return "color: #e2e8f0;"; // Gray for same
    }

    private string GetChangeText(decimal? currentValue, decimal? comparisonValue)
    {
        if (currentValue == null || comparisonValue == null)
            return "";

        var difference = currentValue.Value - comparisonValue.Value;
        if (difference > 0)
            return $"+{difference}";
        else if (difference < 0)
            return difference.ToString();
        else
            return "same";
    }

    private async Task SaveToCacheAsync<T>(string cacheKey, T data) where T : class
    {
        try
        {
            var cacheEntry = new CacheEntry<T>
            {
                Data = data,
                Expiration = DateTime.UtcNow.Add(CacheDuration)
            };
            var json = JsonSerializer.Serialize(cacheEntry);
            await JSRuntime.InvokeVoidAsync("localStorage.setItem", cacheKey, json);
        }
        catch (Exception)
        {
            // Log error if needed
        }
    }

    private class CacheEntry<T>
    {
        public T Data { get; set; }
        public DateTime Expiration { get; set; }
    }
